    83→pattern HandleFactoryStep(order_id: String, input: String, chat_id: String) -> String {
    84→  let step = kv_get("order_step_" + order_id)
    85→  let step_num = to_float(step)
    86→
    87→  if step_num == 1.0 then {
    88→    kv_set("order_step_" + order_id, "2")
    89→    kv_set("order_client_" + order_id, input)
    90→    db_execute("UPDATE factory_orders SET client_name = '" + escape_json(input) + "' WHERE id = '" + order_id + "'")
    91→    return "Step 2/7: What tasks should the office handle?"
    92→  }
    93→  if step_num == 2.0 then {
    94→    kv_set("order_step_" + order_id, "3")
    95→    kv_set("order_tasks_" + order_id, input)
    96→    db_execute("UPDATE factory_orders SET requirements = '" + escape_json(input) + "' WHERE id = '" + order_id + "'")
    97→    return "Step 3/7: Which departments? (yana, osp, lz, expert, marketing, finance, legal, dev, design, qa, visual, engineering, kavalnya)"
    98→  }
    99→  if step_num == 3.0 then {
   100→    kv_set("order_step_" + order_id, "4")
   101→    kv_set("order_depts_" + order_id, input)
   102→    db_execute("UPDATE factory_orders SET departments = '" + escape_json(input) + "' WHERE id = '" + order_id + "'")
   103→    return "Step 4/7: Branding? (name, language, style preferences)"
   104→  }
   105→  if step_num == 4.0 then {
   106→    kv_set("order_step_" + order_id, "5")
   107→    kv_set("order_branding_" + order_id, input)
   108→    db_execute("UPDATE factory_orders SET branding = '" + escape_json(input) + "' WHERE id = '" + order_id + "'")
   109→    return "Step 5/7: Which LLM providers/keys are available?"
   110→  }
   111→  if step_num == 5.0 then {
   112→    kv_set("order_step_" + order_id, "6")
   113→    kv_set("order_llm_" + order_id, input)
   114→    db_execute("UPDATE factory_orders SET llm_config = '" + escape_json(input) + "' WHERE id = '" + order_id + "'")
   115→    let summary = "Order summary:\n"
   116→    let summary = summary + "Client: " + kv_get("order_client_" + order_id) + "\n"
   117→    let summary = summary + "Tasks: " + kv_get("order_tasks_" + order_id) + "\n"
   118→    let summary = summary + "Departments: " + kv_get("order_depts_" + order_id) + "\n"
   119→    let summary = summary + "Branding: " + kv_get("order_branding_" + order_id) + "\n"
   120→    let summary = summary + "LLM: " + input + "\n\nConfirm? Send 'yes' or 'no'."
   121→    kv_set("order_summary_" + order_id, summary)
   122→    return summary
   123→  }
   124→  if step_num == 6.0 then {
   125→    if lower(input) == "yes" or lower(input) == "ok" then {
   126→      kv_set("order_step_" + order_id, "7")
   127→      db_execute("UPDATE factory_orders SET status = 'building' WHERE id = '" + order_id + "'")
   128→      let _ = AuditLog("factory_build", "factory", "building: " + order_id)
   129→      kv_set("active_order", "")
   130→      return "Order confirmed. Building office...\nStatus: building\nTrack: /factory_list"
   131→    } else {
   132→      kv_set("active_order", "")
   133→      db_execute("UPDATE factory_orders SET status = 'cancelled' WHERE id = '" + order_id + "'")
   134→      return "Order cancelled."
   135→    }
   136→  }
   137→  return "Factory flow completed."
   138→}

   140→pattern LZDeepScan(query: String, chat_id: String) -> String {
   141→  let _ = AuditLog("lz_deep_scan", "lz", "query: " + substring(query, 0.0, 100.0))
   142→  let context = BuildContext("lz", "Deep research: " + query, chat_id)
   143→  let messages = json_encode([{role: "system", content: context + "\n\nIMPORTANT: This is a DEEP SCAN request. Use all available sources. Structure findings with [FACT verified] badges."}, {role: "user", content: query}])
   144→  let proxy_result = CallLLMProxy("lz", messages)
   145→  let reply = json_get(parse_json(proxy_result), "reply", "Deep scan failed.")
   146→  let _ = RecordAssistantMessage(chat_id, "lz", reply)
   147→  return reply
   148→}

   182→pattern HandleDeptQuery(dept: String, query: String, chat_id: String) -> String {
   183→  let context = BuildContext(dept, query, chat_id)
   184→  let messages = json_encode([{role: "system", content: context}, {role: "user", content: query}])
   185→  let proxy_result = CallLLMProxy(dept, messages)
   186→  let reply = json_get(parse_json(proxy_result), "reply", "Error: empty reply")
   187→  let _ = RecordAssistantMessage(chat_id, dept, reply)
   188→  let _ = AuditLog("dept_query", dept, "query: " + substring(query, 0.0, 100.0))
   189→  return reply
   190→}

   192→pattern DetectDepartment(text: String) -> String {
   193→  if starts_with(text, "/yana") then { return "yana" }
   194→  if starts_with(text, "/osp") then { return "osp" }
   195→  if starts_with(text, "/lz") then { return "lz" }

   198→  if starts_with(text, "/finance") then { return "finance" }
   199→  if starts_with(text, "/legal") then { return "legal" }
   200→  if starts_with(text, "/dev") then { return "dev" }
   201→  if starts_with(text, "/design") then { return "design" }
   202→  if starts_with(text, "/qa") then { return "qa" }
   203→  if starts_with(text, "/visual") then { return "visual" }
   204→  if starts_with(text, "/engineering") then { return "engineering" }
   205→  if starts_with(text, "/forge") then { return "kavalnya" }
   206→  if starts_with(text, "/admin") then { return "admin" }

   209→  if starts_with(text, "/analyze") then { return "osp" }
   210→  if starts_with(text, "/scan") then { return "lz" }

   212→  if starts_with(text, "/quickanalyze") then { return "osp" }
   213→  if starts_with(text, "/wargame") then { return "osp" }
   214→  if starts_with(text, "/market-research") then { return "marketing" }
   215→  if starts_with(text, "/segment") then { return "marketing" }
   216→  if starts_with(text, "/competitors") then { return "marketing" }
   217→  if starts_with(text, "/campaign") then { return "marketing" }
   218→  if starts_with(text, "/calc") then { return "engineering" }
   219→  if starts_with(text, "/simulate") then { return "engineering" }
   220→
   221→  // Russian keyword detection via proxy
   222→  let lower_text = lower(text)

   225→  let detected = json_get(parse_json(result), "dept", "")
   226→  if detected != "" then { return detected }
   227→
   228→  return ""
