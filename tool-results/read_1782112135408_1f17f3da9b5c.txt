    83→    83→pattern HandleFactoryStep(order_id: String, input: String, chat_id: String) -> String {
    84→    84→  let step = kv_get("order_step_" + order_id)
    85→    85→  let step_num = to_float(step)
    86→    86→
    87→    87→  if step_num == 1.0 then {
    88→    88→    kv_set("order_step_" + order_id, "2")
    89→    89→    kv_set("order_client_" + order_id, input)
    90→    90→    db_execute("UPDATE factory_orders SET client_name = '" + escape_json(input) + "' WHERE id = '" + order_id + "'")
    91→    91→    return "Step 2/7: What tasks should the office handle?"
    92→    92→  }
    93→    93→  if step_num == 2.0 then {
    94→    94→    kv_set("order_step_" + order_id, "3")
    95→    95→    kv_set("order_tasks_" + order_id, input)
    96→    96→    db_execute("UPDATE factory_orders SET requirements = '" + escape_json(input) + "' WHERE id = '" + order_id + "'")
    97→    97→    return "Step 3/7: Which departments? (yana, osp, lz, expert, marketing, finance, legal, dev, design, qa, visual, engineering, kavalnya)"
    98→    98→  }
    99→    99→  if step_num == 3.0 then {
   100→   100→    kv_set("order_step_" + order_id, "4")
   101→   101→    kv_set("order_depts_" + order_id, input)
   102→   102→    db_execute("UPDATE factory_orders SET departments = '" + escape_json(input) + "' WHERE id = '" + order_id + "'")
   103→   103→    return "Step 4/7: Branding? (name, language, style preferences)"
   104→   104→  }
   105→   105→  if step_num == 4.0 then {
   106→   106→    kv_set("order_step_" + order_id, "5")
   107→   107→    kv_set("order_branding_" + order_id, input)
   108→   108→    db_execute("UPDATE factory_orders SET branding = '" + escape_json(input) + "' WHERE id = '" + order_id + "'")
   109→   109→    return "Step 5/7: Which LLM providers/keys are available?"
   110→   110→  }
   111→   111→  if step_num == 5.0 then {
   112→   112→    kv_set("order_step_" + order_id, "6")
   113→   113→    kv_set("order_llm_" + order_id, input)
   114→   114→    db_execute("UPDATE factory_orders SET llm_config = '" + escape_json(input) + "' WHERE id = '" + order_id + "'")
   115→   115→    let summary = "Order summary:\n"
   116→   116→    let summary = summary + "Client: " + kv_get("order_client_" + order_id) + "\n"
   117→   117→    let summary = summary + "Tasks: " + kv_get("order_tasks_" + order_id) + "\n"
   118→   118→    let summary = summary + "Departments: " + kv_get("order_depts_" + order_id) + "\n"
   119→   119→    let summary = summary + "Branding: " + kv_get("order_branding_" + order_id) + "\n"
   120→   120→    let summary = summary + "LLM: " + input + "\n\nConfirm? Send 'yes' or 'no'."
   121→   121→    kv_set("order_summary_" + order_id, summary)
   122→   122→    return summary
   123→   123→  }
   124→   124→  if step_num == 6.0 then {
   125→   125→    if lower(input) == "yes" or lower(input) == "ok" then {
   126→   126→      kv_set("order_step_" + order_id, "7")
   127→   127→      db_execute("UPDATE factory_orders SET status = 'building' WHERE id = '" + order_id + "'")
   128→   128→      let _ = AuditLog("factory_build", "factory", "building: " + order_id)
   129→   129→      kv_set("active_order", "")
   130→   130→      return "Order confirmed. Building office...\nStatus: building\nTrack: /factory_list"
   131→   131→    } else {
   132→   132→      kv_set("active_order", "")
   133→   133→      db_execute("UPDATE factory_orders SET status = 'cancelled' WHERE id = '" + order_id + "'")
   134→   134→      return "Order cancelled."
   135→   135→    }
   136→   136→  }
   137→   137→  return "Factory flow completed."
   138→   138→}

   141→   141→  // 9-source deep research via Apodex
   142→   142→  let _ = AuditLog("lz_deep_scan", "lz", "query: " + substring(query, 0.0, 100.0))
   143→   143→  let context = BuildContext("lz", "Deep research: " + query, chat_id)
   144→   144→  let messages = json_encode([{role: "system", content: context + "\n\nIMPORTANT: This is a DEEP SCAN request. Use all available sources. Structure findings with [FACT verified] badges."}, {role: "user", content: query}])

   149→   149→  return reply
