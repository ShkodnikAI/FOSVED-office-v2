   221→   221→# ──────────────────────────── HTML STRIP FOR ARCHIVE ────────────────────────────
   222→   222→def strip_html_for_archive(html_content):
   223→   223→    """Convert HTML badges to text markers and strip all tags for Telegram."""
   224→   224→    text = html_content
   225→   225→    # Convert badge spans with known classes to text markers
   226→   226→    badge_map = {
   227→   227→        'badge-fact': '[FACT]',
   228→   228→        'badge-int': '[INTERPRETATION]',
   229→   229→        'badge-spec': '[SPECULATION]',
   230→   230→    }
   231→   231→    for cls, marker in badge_map.items():
   232→   232→        # Match <span class="badge badge-xxx">content</span>
   233→   233→        pattern = r'<span[^>]*class="[^"]*\b' + re.escape(cls) + r'\b[^"]*"[^>]*>(.*?)</span>'
   234→   234→        text = re.sub(pattern, marker + r' \1 ', text, flags=re.IGNORECASE | re.DOTALL)
   235→   235→    # Also match class="badge badge-xxx" without space (single class)
   236→   236→    for cls, marker in badge_map.items():
   237→   237→        pattern2 = r'<span[^>]*class="' + re.escape(cls) + r'"[^>]*>(.*?)</span>'
   238→   238→        text = re.sub(pattern2, marker + r' \1 ', text, flags=re.IGNORECASE | re.DOTALL)
   239→   239→    # Remove all remaining HTML tags
   240→   240→    text = re.sub(r'<br\s*/?>', '\n', text, flags=re.IGNORECASE)
   241→   241→    text = re.sub(r'</p>', '\n', text, flags=re.IGNORECASE)
   242→   242→    text = re.sub(r'</div>', '\n', text, flags=re.IGNORECASE)
   243→   243→    text = re.sub(r'</li>', '\n- ', text, flags=re.IGNORECASE)
   244→   244→    text = re.sub(r'<li[^>]*>', '- ', text, flags=re.IGNORECASE)
   245→   245→    text = re.sub(r'<[^>]+>', ' ', text)
   246→   246→    # Collapse whitespace
   247→   247→    text = re.sub(r'[ \t]+', ' ', text)
   248→   248→    text = re.sub(r'\n[ \t]+', '\n', text)
   249→   249→    text = re.sub(r'\n{3,}', '\n\n', text)
   250→   250→    return text.strip()

   252→   252→# ──────────────────────────── LLM CALL ────────────────────────────
   253→   253→def call_llm(dept, messages, model_override=None):
   254→   254→    priority = _cfg_cache["priority_provider"]
   255→   255→    model = model_override or _cfg_cache["overrides"].get(dept) or _cfg_cache["priority_model"] or "glm-4.6"
   256→   256→    order = [p for p in [priority] + [k for k in PROVIDERS if k != priority] if p in PROVIDERS]
   257→   257→    last_err = None
   258→   258→    for prov_name in order:
   259→   259→        prov = PROVIDERS[prov_name]
   260→   260→        api_key = os.environ.get(prov["env_key"])
   261→   261→        if not api_key:
   262→   262→            continue
   263→   263→        try:
   264→   264→            use_model = model
   265→   265→            if prov_name == "anthropic" and use_model.startswith("gpt"):
   266→   266→                use_model = "claude-sonnet-4-20250514"
   267→   267→            elif prov_name == "deepseek" and not use_model.startswith("deepseek"):
   268→   268→                use_model = "deepseek-chat"
   269→   269→            elif prov_name == "groq" and not any(m in use_model for m in ("llama", "mixtral", "gemma")):
   270→   270→                use_model = "llama-3.3-70b-versatile"
   271→   271→
   272→   272→            if prov["format"] == "openai":
   273→   273→                headers = {"Authorization": f"Bearer {api_key}", "Content-Type": "application/json"}
   274→   274→                body = {"model": use_model, "messages": messages, "temperature": 0.7}
   275→   275→            else:
   276→   276→                sys_msg = ""
   277→   277→                user_msgs = []
   278→   278→                for m in messages:
   279→   279→                    if m["role"] == "system": sys_msg = m["content"]
   280→   280→                    else: user_msgs.append(m)
   281→   281→                headers = {"x-api-key": api_key, "anthropic-version": "2023-06-01", "Content-Type": "application/json"}
   282→   282→                body = {"model": use_model, "max_tokens": 4096, "system": sys_msg, "messages": user_msgs}
   283→   283→
   284→   284→            resp = http_requests.post(prov["url"], headers=headers, json=body, timeout=prov["timeout"])
   285→   285→            resp.raise_for_status()
   286→   286→            data = resp.json()
   287→   287→
   288→   288→            if prov["format"] == "openai":
   289→   289→                reply = data["choices"][0]["message"]["content"]
   290→   290→                t_in = data.get("usage", {}).get("prompt_tokens", 0)
   291→   291→                t_out = data.get("usage", {}).get("completion_tokens", 0)
   292→   292→            else:
   293→   293→                reply = data["content"][0]["text"]
   294→   294→                t_in = data.get("usage", {}).get("input_tokens", 0)
   295→   295→                t_out = data.get("usage", {}).get("output_tokens", 0)
   296→   296→
   297→   297→            costs = TOKEN_COSTS.get(prov_name, {"input": 1, "output": 1})
   298→   298→            cost = t_in * costs["input"] / 1_000_000 + t_out * costs["output"] / 1_000_000
   299→   299→            now = datetime.utcnow().isoformat()
   300→   300→            _query("INSERT INTO token_usage (provider,model,tokens_in,tokens_out,cost,dept,created) VALUES (?,?,?,?,?,?,?)",
   301→   301→                   (prov_name, use_model, t_in, t_out, round(cost, 6), dept, now))
   302→   302→            _query("INSERT INTO audit_log (action,details,created) VALUES (?,?,?)",
   303→   303→                   ("llm_call", json.dumps({"dept": dept, "provider": prov_name, "model": use_model, "tokens_in": t_in, "tokens_out": t_out}), now))
   304→   304→            log.info(f"LLM ok: provider={prov_name} model={use_model} dept={dept} t_in={t_in} t_out={t_out}")
   305→   305→            return {"reply": reply, "provider": prov_name, "model": use_model, "tokens_in": t_in, "tokens_out": t_out}
   306→   306→        except Exception as e:
   307→   307→            last_err = str(e)
   308→   308→            log.warning(f"LLM fail: provider={prov_name} err={last_err}")
   309→   309→    return {"reply": f"[Oshibka] Vse providery nedostupny: {last_err}", "provider": None, "model": None, "tokens_in": 0, "tokens_out": 0}

   311→   311→# ──────────────────────────── TELEGRAM HELPERS ────────────────────────────
   312→   312→def _tg_send_message(chat_id, text, parse_mode=None):

   851→   851→def _collect_metrics_for_period(start_iso, end_iso=None):
   852→   852→    """Collect metrics for a given time period."""
   853→   853→    conditions = "created >= ?"
   854→   854→    params = [start_iso]
   855→   855→    if end_iso:
   856→   856→        conditions += " AND created < ?"
   857→   857→        params.append(end_iso)
   858→   858→
   859→   859→    tasks = _query(f"SELECT dept, COUNT(*) as c FROM dialog_history WHERE {conditions} AND role='user' GROUP BY dept", params, fetch=True)
   860→   860→    reports = _query(f"SELECT dept, COUNT(*) as c FROM miniapp_responses WHERE {conditions} GROUP BY dept", params, fetch=True)
   861→   861→    corrections = _query(f"SELECT dept, COUNT(*) as c FROM adapt_corrections WHERE {conditions} GROUP BY dept", params, fetch=True)
   862→   862→    ratings = _query(f"SELECT dept, COALESCE(AVG(rating),0) as avg_r, COUNT(*) as c FROM dept_metrics WHERE {conditions} GROUP BY dept", params, fetch=True)
   863→   863→    tokens = _query(f"SELECT dept, COALESCE(SUM(tokens_in),0) as ti, COALESCE(SUM(tokens_out),0) as to2, COALESCE(SUM(cost),0) as cost FROM token_usage WHERE {conditions} GROUP BY dept", params, fetch=True)
   864→   864→
   865→   865→    return {
   866→   866→        "tasks": {r["dept"]: r["c"] for r in tasks},
   867→   867→        "reports": {r["dept"]: r["c"] for r in reports},
   868→   868→        "corrections": {r["dept"]: r["c"] for r in corrections},
   869→   869→        "ratings": {r["dept"]: {"avg": r["avg_r"], "count": r["c"]} for r in ratings},
   870→   870→        "tokens": {r["dept"]: {"in": r["ti"], "out": r["to2"], "cost": r["cost"]} for r in tokens},
   871→   871→    }
   872→   872→
   873→   873→def _format_metrics_text(metrics):
   874→   874→    """Format metrics dict as readable text for LLM prompt."""
   875→   875→    lines = []
   876→   876→    lines.append("METRICS:")
   877→   877→    all_depts = set(list(metrics["tasks"].keys()) + list(metrics["reports"].keys()) + list(metrics["corrections"].keys()) + list(metrics["ratings"].keys()))
   878→   878→    for dept in sorted(all_depts):
   879→   879→        name = DEPT_NAMES.get(dept, dept)
   880→   880→        parts = [f"{name}:"]
   881→   881→        if dept in metrics["tasks"]:
   882→   882→            parts.append(f"tasks={metrics['tasks'][dept]}")
   883→   883→        if dept in metrics["reports"]:
   884→   884→            parts.append(f"reports={metrics['reports'][dept]}")
   885→   885→        if dept in metrics["corrections"]:
   886→   886→            parts.append(f"corrections={metrics['corrections'][dept]}")
   887→   887→        if dept in metrics["ratings"]:
   888→   888→            r = metrics["ratings"][dept]
   889→   889→            parts.append(f"avg_rating={r['avg']:.1f} (n={r['count']})")
   890→   890→        if dept in metrics["tokens"]:
   891→   891→            t = metrics["tokens"][dept]
   892→   892→            parts.append(f"tokens={t['in']+t['out']} cost=${t['cost']:.2f}")
   893→   893→        lines.append("  " + " ".join(parts))
   894→   894→    return "\n".join(lines)
