   124→def detect_dept(text):

   126→    for cmd, dept in sorted(COMMAND_DEPT.items(), key=lambda x: -len(x[0])):
   127→        if text_lower.startswith(cmd) and (len(text_lower) == len(cmd) or text_lower[len(cmd)] in ' \n/'):
   128→            return dept
   129→    for dept, kws in RUSSIAN_KEYWORDS.items():
   130→        for kw in kws:
   131→            if kw in text_lower:
   132→                return dept
   133→    return 'yana'

   208→def _query(sql, params=(), fetch=False):
   209→    with db_lock:
   210→        c = _db()
   211→        cur = c.execute(sql, params)
   212→        if fetch:
   213→            rows = cur.fetchall(); c.close(); return rows
   214→        c.commit(); c.close()

   221→# ──────────────────────────── HTML STRIP FOR ARCHIVE ────────────────────────────
   222→def strip_html_for_archive(html_content):
   223→    """Convert HTML badges to text markers and strip all tags for Telegram."""
   224→    text = html_content
   225→    # Convert badge spans with known classes to text markers
   226→    badge_map = {
   227→        'badge-fact': '[FACT]',
   228→        'badge-int': '[INTERPRETATION]',
   229→        'badge-spec': '[SPECULATION]',
   230→    }
   231→    for cls, marker in badge_map.items():
   232→        # Match <span class="badge badge-xxx">content</span>
   233→        pattern = r'<span[^>]*class="[^"]*\b' + re.escape(cls) + r'\b[^"]*"[^>]*>(.*?)</span>'
   234→        text = re.sub(pattern, marker + r' \1 ', text, flags=re.IGNORECASE | re.DOTALL)
   235→    # Also match class="badge badge-xxx" without space (single class)
   236→    for cls, marker in badge_map.items():
   237→        pattern2 = r'<span[^>]*class="' + re.escape(cls) + r'"[^>]*>(.*?)</span>'
   238→        text = re.sub(pattern2, marker + r' \1 ', text, flags=re.IGNORECASE | re.DOTALL)
   239→    # Remove all remaining HTML tags
   240→    text = re.sub(r'<br\s*/?>', '\n', text, flags=re.IGNORECASE)
   241→    text = re.sub(r'</p>', '\n', text, flags=re.IGNORECASE)
   242→    text = re.sub(r'</div>', '\n', text, flags=re.IGNORECASE)
   243→    text = re.sub(r'</li>', '\n- ', text, flags=re.IGNORECASE)
   244→    text = re.sub(r'<li[^>]*>', '- ', text, flags=re.IGNORECASE)
   245→    text = re.sub(r'<[^>]+>', ' ', text)
   246→    # Collapse whitespace
   247→    text = re.sub(r'[ \t]+', ' ', text)
   248→    text = re.sub(r'\n[ \t]+', '\n', text)
   249→    text = re.sub(r'\n{3,}', '\n\n', text)
   250→    return text.strip()

   252→# ──────────────────────────── LLM CALL ────────────────────────────
   253→def call_llm(dept, messages, model_override=None):
   254→    priority = _cfg_cache["priority_provider"]
   255→    model = model_override or _cfg_cache["overrides"].get(dept) or _cfg_cache["priority_model"] or "glm-4.6"
   256→    order = [p for p in [priority] + [k for k in PROVIDERS if k != priority] if p in PROVIDERS]
   257→    last_err = None
   258→    for prov_name in order:
   259→        prov = PROVIDERS[prov_name]
   260→        api_key = os.environ.get(prov["env_key"])
   261→        if not api_key:
   262→            continue
   263→        try:
   264→            use_model = model
   265→            if prov_name == "anthropic" and use_model.startswith("gpt"):
   266→                use_model = "claude-sonnet-4-20250514"
   267→            elif prov_name == "deepseek" and not use_model.startswith("deepseek"):
   268→                use_model = "deepseek-chat"
   269→            elif prov_name == "groq" and not any(m in use_model for m in ("llama", "mixtral", "gemma")):
   270→                use_model = "llama-3.3-70b-versatile"
   271→
   272→            if prov["format"] == "openai":
   273→                headers = {"Authorization": f"Bearer {api_key}", "Content-Type": "application/json"}
   274→                body = {"model": use_model, "messages": messages, "temperature": 0.7}
   275→            else:
   276→                sys_msg = ""
   277→                user_msgs = []
   278→                for m in messages:
   279→                    if m["role"] == "system": sys_msg = m["content"]
   280→                    else: user_msgs.append(m)
   281→                headers = {"x-api-key": api_key, "anthropic-version": "2023-06-01", "Content-Type": "application/json"}
   282→                body = {"model": use_model, "max_tokens": 4096, "system": sys_msg, "messages": user_msgs}
   283→
   284→            resp = http_requests.post(prov["url"], headers=headers, json=body, timeout=prov["timeout"])
   285→            resp.raise_for_status()
   286→            data = resp.json()
   287→
   288→            if prov["format"] == "openai":
   289→                reply = data["choices"][0]["message"]["content"]
   290→                t_in = data.get("usage", {}).get("prompt_tokens", 0)
   291→                t_out = data.get("usage", {}).get("completion_tokens", 0)
   292→            else:
   293→                reply = data["content"][0]["text"]
   294→                t_in = data.get("usage", {}).get("input_tokens", 0)
   295→                t_out = data.get("usage", {}).get("output_tokens", 0)
   296→
   297→            costs = TOKEN_COSTS.get(prov_name, {"input": 1, "output": 1})
   298→            cost = t_in * costs["input"] / 1_000_000 + t_out * costs["output"] / 1_000_000
   299→            now = datetime.utcnow().isoformat()
   300→            _query("INSERT INTO token_usage (provider,model,tokens_in,tokens_out,cost,dept,created) VALUES (?,?,?,?,?,?,?)",
   301→                   (prov_name, use_model, t_in, t_out, round(cost, 6), dept, now))
   302→            _query("INSERT INTO audit_log (action,details,created) VALUES (?,?,?)",
   303→                   ("llm_call", json.dumps({"dept": dept, "provider": prov_name, "model": use_model, "tokens_in": t_in, "tokens_out": t_out}), now))
   304→            log.info(f"LLM ok: provider={prov_name} model={use_model} dept={dept} t_in={t_in} t_out={t_out}")
   305→            return {"reply": reply, "provider": prov_name, "model": use_model, "tokens_in": t_in, "tokens_out": t_out}
   306→        except Exception as e:
   307→            last_err = str(e)
   308→            log.warning(f"LLM fail: provider={prov_name} err={last_err}")
   309→    return {"reply": f"[Oshibka] Vse providery nedostupny: {last_err}", "provider": None, "model": None, "tokens_in": 0, "tokens_out": 0}

   311→# ──────────────────────────── TELEGRAM HELPERS ────────────────────────────
   312→def _tg_send_message(chat_id, text, parse_mode=None):
   313→    """Send a text message via Telegram Bot API."""
   314→    if not TELEGRAM_BOT_TOKEN:
   315→        log.warning("TELEGRAM_BOT_TOKEN not set, cannot send message")
   316→        return None
   317→    url = f"https://api.telegram.org/bot{TELEGRAM_BOT_TOKEN}/sendMessage"
   318→    payload = {"chat_id": chat_id, "text": text}
   319→    if parse_mode:
   320→        payload["parse_mode"] = parse_mode
   321→    try:
   322→        resp = http_requests.post(url, json=payload, timeout=30)
   323→        resp.raise_for_status()
   324→        data = resp.json()
   325→        if data.get("ok"):
   326→            return data["result"].get("message_id")
   327→    except Exception as e:
   328→        log.error(f"Telegram sendMessage failed: {e}")
   329→    return None

   331→def _tg_send_voice(chat_id, voice_path):
   332→    """Send a voice file via Telegram Bot API using multipart upload."""
   333→    if not TELEGRAM_BOT_TOKEN:
   334→        log.warning("TELEGRAM_BOT_TOKEN not set, cannot send voice")
   335→        return None
   336→    url = f"https://api.telegram.org/bot{TELEGRAM_BOT_TOKEN}/sendVoice"
   337→    try:
   338→        with open(voice_path, "rb") as f:
   339→            resp = http_requests.post(url, data={"chat_id": chat_id}, files={"voice": f}, timeout=60)
   340→        resp.raise_for_status()
   341→        data = resp.json()
   342→        if data.get("ok"):
   343→            return data["result"].get("message_id")
   344→    except Exception as e:
   345→        log.error(f"Telegram sendVoice failed: {e}")
   346→    return None

   369→    """Get file path from Telegram Bot API given a file_id."""
   370→    if not TELEGRAM_BOT_TOKEN:
   371→        return None
   372→    url = f"https://api.telegram.org/bot{TELEGRAM_BOT_TOKEN}/getFile"
   373→    try:
   374→        resp = http_requests.get(url, params={"file_id": file_id}, timeout=15)
   375→        resp.raise_for_status()
   376→        data = resp.json()
   377→        if data.get("ok") and data.get("result", {}).get("file_path"):
   378→            return data["result"]["file_path"]
   379→    except Exception as e:
   380→        log.error(f"Telegram getFile failed: {e}")
   381→    return None

   387→    url = f"https://api.telegram.org/file/bot{TELEGRAM_BOT_TOKEN}/{file_path}"
   388→    try:
   389→        resp = http_requests.get(url, timeout=60)
   390→        resp.raise_for_status()
   391→        return resp.content
   392→    except Exception as e:

   609→@app.post("/tts")
   610→def tts_endpoint():
   611→    """Text-to-speech via edge-tts, send voice to Telegram."""
   612→    d = request.get_json(force=True)
   613→    text = d.get("text", "")
   614→    voice = d.get("voice", "ru-RU-DmitryNeural")
   615→    chat_id = d.get("chat_id", "")
   616→
   617→    if not text:
   618→        return jsonify({"status": "error", "message": "text is required"}), 400
   619→    if not chat_id:
   620→        return jsonify({"status": "error", "message": "chat_id is required"}), 400
   621→
   622→    timestamp = int(time.time())
   623→    tmp_path = f"/tmp/tts_{timestamp}_{uuid.uuid4().hex[:6]}.mp3"
   624→
   625→    try:
   626→        # Run edge-tts subprocess
   627→        cmd = [
   628→            "edge-tts",
   629→            "--voice", voice,
   630→            "--text", text,
   631→            "--write-media", tmp_path
   632→        ]
   633→        result = subprocess.run(cmd, capture_output=True, text=True, timeout=60)
   634→        if result.returncode != 0:
   635→            log.error(f"edge-tts failed: {result.stderr}")
   636→            _log_audit("tts_error", {"voice": voice, "error": result.stderr[:500]})
   637→            return jsonify({"status": "error", "message": "edge-tts synthesis failed", "detail": result.stderr[:500]}), 500
   638→
   639→        # Send voice to Telegram
   640→        msg_id = _tg_send_voice(chat_id, tmp_path)
   641→
   642→        _log_audit("tts", {"chat_id": chat_id, "voice": voice, "text_len": len(text), "tg_msg_id": msg_id})
   643→
   644→        return jsonify({"status": "ok", "msg_id": msg_id})
   645→
   646→    except FileNotFoundError:
   647→        _log_audit("tts_error", {"error": "edge-tts not installed"})
   648→        return jsonify({"status": "error", "message": "edge-tts binary not found"}), 500
   649→    except subprocess.TimeoutExpired:
   650→        _log_audit("tts_error", {"error": "timeout"})
   651→        return jsonify({"status": "error", "message": "TTS synthesis timeout"}), 500
   652→    except Exception as e:
   653→        log.error(f"TTS error: {e}")
   654→        _log_audit("tts_error", {"error": str(e)[:500]})
   655→        return jsonify({"status": "error", "message": str(e)}), 500
   656→    finally:
   657→        # Clean up temp file
   658→        try:
   659→            if os.path.exists(tmp_path):
   660→                os.remove(tmp_path)
   661→        except Exception:
   662→            pass

   667→
   668→@app.post("/stt")
   669→def stt_endpoint():
   670→    """Speech-to-text via Whisper (Groq preferred, OpenAI fallback)."""
   671→    d = request.get_json(force=True)
   672→    file_id = d.get("file_id", "")
   673→
   674→    if not file_id:
   675→        return jsonify({"status": "error", "message": "file_id is required"}), 400
   676→
   677→    provider = WHISPER_PROVIDER
   678→    audio_data = None
   679→
   680→    try:
   681→        # Step 1: Get file path from Telegram
   682→        file_path = _tg_get_file(file_id)
   683→        if not file_path:
   684→            _log_audit("stt_error", {"error": "could not get file path from Telegram", "file_id": file_id})
   685→            return jsonify({"status": "error", "message": "Failed to get file from Telegram"}), 400
   686→
   687→        # Step 2: Download audio from Telegram
   688→        audio_data = _tg_download_file(file_path)
   689→        if not audio_data:
   690→            _log_audit("stt_error", {"error": "could not download file from Telegram", "file_path": file_path})
   691→            return jsonify({"status": "error", "message": "Failed to download audio from Telegram"}), 400
   692→
   693→        # Step 3: Transcribe via Whisper
   694→        if provider == "groq":
   695→            api_key = os.environ.get("GROQ_API_KEY", WHISPER_API_KEY)
   696→            url = "https://api.groq.com/openai/v1/audio/transcriptions"
   697→            model = "whisper-large-v3"
   698→        else:
   699→            api_key = os.environ.get("OPENAI_API_KEY", WHISPER_API_KEY)
   700→            url = "https://api.openai.com/v1/audio/transcriptions"
   701→            model = "whisper-1"
   702→
   703→        # Prepare multipart form
   704→        files = {
   705→            "file": ("audio.ogg", bytes(audio_data), "audio/ogg"),
   706→        }
   707→        data = {"model": model}
   708→        headers = {"Authorization": f"Bearer {api_key}"}
   709→
   710→        resp = http_requests.post(url, headers=headers, files=files, data=data, timeout=60)
   711→        resp.raise_for_status()
   712→        resp_data = resp.json()
   713→        transcribed_text = resp_data.get("text", "")
   714→
   715→        _log_audit("stt", {"file_id": file_id, "provider": provider, "model": model, "text_len": len(transcribed_text)})
   716→
   717→        return jsonify({"status": "ok", "text": transcribed_text, "provider": provider})
   718→
   719→    except Exception as e:
   720→        log.error(f"STT error (provider={provider}): {e}")
   721→        # Try fallback provider if primary failed
   722→        if provider == "groq" and audio_data:
   723→            try:
   724→                fallback_key = os.environ.get("OPENAI_API_KEY", "")
   725→                if not fallback_key:
   726→                    raise ValueError("No OpenAI API key for fallback")
   727→                files = {"file": ("audio.ogg", bytes(audio_data), "audio/ogg")}
   728→                data = {"model": "whisper-1"}

   851→def _collect_metrics_for_period(start_iso, end_iso=None):
   852→    """Collect metrics for a given time period."""
   853→    conditions = "created >= ?"
   854→    params = [start_iso]
   855→    if end_iso:
   856→        conditions += " AND created < ?"
   857→        params.append(end_iso)
   858→
   859→    tasks = _query(f"SELECT dept, COUNT(*) as c FROM dialog_history WHERE {conditions} AND role='user' GROUP BY dept", params, fetch=True)
   860→    reports = _query(f"SELECT dept, COUNT(*) as c FROM miniapp_responses WHERE {conditions} GROUP BY dept", params, fetch=True)
   861→    corrections = _query(f"SELECT dept, COUNT(*) as c FROM adapt_corrections WHERE {conditions} GROUP BY dept", params, fetch=True)
   862→    ratings = _query(f"SELECT dept, COALESCE(AVG(rating),0) as avg_r, COUNT(*) as c FROM dept_metrics WHERE {conditions} GROUP BY dept", params, fetch=True)
   863→    tokens = _query(f"SELECT dept, COALESCE(SUM(tokens_in),0) as ti, COALESCE(SUM(tokens_out),0) as to2, COALESCE(SUM(cost),0) as cost FROM token_usage WHERE {conditions} GROUP BY dept", params, fetch=True)
   864→
   865→    return {
   866→        "tasks": {r["dept"]: r["c"] for r in tasks},
   867→        "reports": {r["dept"]: r["c"] for r in reports},
   868→        "corrections": {r["dept"]: r["c"] for r in corrections},
   869→        "ratings": {r["dept"]: {"avg": r["avg_r"], "count": r["c"]} for r in ratings},
   870→        "tokens": {r["dept"]: {"in": r["ti"], "out": r["to2"], "cost": r["cost"]} for r in tokens},
   871→    }
   872→
   873→def _format_metrics_text(metrics):
   874→    """Format metrics dict as readable text for LLM prompt."""
   875→    lines = []
   876→    lines.append("METRICS:")
   877→    all_depts = set(list(metrics["tasks"].keys()) + list(metrics["reports"].keys()) + list(metrics["corrections"].keys()) + list(metrics["ratings"].keys()))
   878→    for dept in sorted(all_depts):
   879→        name = DEPT_NAMES.get(dept, dept)
   880→        parts = [f"{name}:"]
   881→        if dept in metrics["tasks"]:
   882→            parts.append(f"tasks={metrics['tasks'][dept]}")
   883→        if dept in metrics["reports"]:
   884→            parts.append(f"reports={metrics['reports'][dept]}")
   885→        if dept in metrics["corrections"]:
   886→            parts.append(f"corrections={metrics['corrections'][dept]}")
   887→        if dept in metrics["ratings"]:
   888→            r = metrics["ratings"][dept]
   889→            parts.append(f"avg_rating={r['avg']:.1f} (n={r['count']})")
   890→        if dept in metrics["tokens"]:
   891→            t = metrics["tokens"][dept]
   892→            parts.append(f"tokens={t['in']+t['out']} cost=${t['cost']:.2f}")
   893→        lines.append("  " + " ".join(parts))
   894→    return "\n".join(lines)
