   126→def detect_dept(text):

   128→    for cmd, dept in sorted(COMMAND_DEPT.items(), key=lambda x: -len(x[0])):
   129→        if text_lower.startswith(cmd) and (len(text_lower) == len(cmd) or text_lower[len(cmd)] in ' \n/'):
   130→            return dept
   131→    for dept, kws in RUSSIAN_KEYWORDS.items():
   132→        for kw in kws:
   133→            if kw in text_lower:
   134→                return dept
   135→    return 'yana'

   223→# ──────────────────────────── HTML STRIP FOR ARCHIVE ────────────────────────────
   224→def strip_html_for_archive(html_content):
   225→    """Convert HTML badges to text markers and strip all tags for Telegram."""
   226→    text = html_content
   227→    # Convert badge spans with known classes to text markers
   228→    badge_map = {
   229→        'badge-fact': '[FACT]',
   230→        'badge-int': '[INTERPRETATION]',
   231→        'badge-spec': '[SPECULATION]',
   232→        'badge-verified': '[FACT verified]',
   233→        'badge-unverified': '[FACT unverified]',
   234→        'badge-conflicting': '[CONFLICT]',
   235→    }
   236→    for cls, marker in badge_map.items():
   237→        # Match <span class="badge badge-xxx">content</span>
   238→        pattern = r'<span[^>]*class="[^"]*\b' + re.escape(cls) + r'\b[^"]*"[^>]*>(.*?)</span>'
   239→        text = re.sub(pattern, marker + r' \1 ', text, flags=re.IGNORECASE | re.DOTALL)
   240→    # Also match class="badge badge-xxx" without space (single class)
   241→    for cls, marker in badge_map.items():
   242→        pattern2 = r'<span[^>]*class="' + re.escape(cls) + r'"[^>]*>(.*?)</span>'
   243→        text = re.sub(pattern2, marker + r' \1 ', text, flags=re.IGNORECASE | re.DOTALL)
   244→    # Remove all remaining HTML tags
   245→    text = re.sub(r'<br\s*/?>', '\n', text, flags=re.IGNORECASE)
   246→    text = re.sub(r'</p>', '\n', text, flags=re.IGNORECASE)
   247→    text = re.sub(r'</div>', '\n', text, flags=re.IGNORECASE)
   248→    text = re.sub(r'</li>', '\n- ', text, flags=re.IGNORECASE)
   249→    text = re.sub(r'<li[^>]*>', '- ', text, flags=re.IGNORECASE)
   250→    text = re.sub(r'<[^>]+>', ' ', text)
   251→    # Collapse whitespace
   252→    text = re.sub(r'[ \t]+', ' ', text)
   253→    text = re.sub(r'\n[ \t]+', '\n', text)
   254→    text = re.sub(r'\n{3,}', '\n\n', text)
   255→    return text.strip()

   257→# ──────────────────────────── LLM CALL ────────────────────────────
   258→def call_llm(dept, messages, model_override=None):
   259→    priority = _cfg_cache["priority_provider"]
   260→    model = model_override or _cfg_cache["overrides"].get(dept) or _cfg_cache["priority_model"] or "glm-4.6"
   261→    order = [p for p in [priority] + [k for k in PROVIDERS if k != priority] if p in PROVIDERS]
   262→    last_err = None
   263→    for prov_name in order:
   264→        prov = PROVIDERS[prov_name]
   265→        api_key = os.environ.get(prov["env_key"])
   266→        if not api_key:
   267→            continue
   268→        try:
   269→            use_model = model
   270→            if prov_name == "anthropic" and use_model.startswith("gpt"):
   271→                use_model = "claude-sonnet-4-20250514"
   272→            elif prov_name == "deepseek" and not use_model.startswith("deepseek"):
   273→                use_model = "deepseek-chat"
   274→            elif prov_name == "groq" and not any(m in use_model for m in ("llama", "mixtral", "gemma")):
   275→                use_model = "llama-3.3-70b-versatile"
   276→
   277→            if prov["format"] == "openai":
   278→                headers = {"Authorization": f"Bearer {api_key}", "Content-Type": "application/json"}
   279→                body = {"model": use_model, "messages": messages, "temperature": 0.7}
   280→            else:
   281→                sys_msg = ""
   282→                user_msgs = []
   283→                for m in messages:
   284→                    if m["role"] == "system": sys_msg = m["content"]
   285→                    else: user_msgs.append(m)
   286→                headers = {"x-api-key": api_key, "anthropic-version": "2023-06-01", "Content-Type": "application/json"}
   287→                body = {"model": use_model, "max_tokens": 4096, "system": sys_msg, "messages": user_msgs}
   288→
   289→            resp = http_requests.post(prov["url"], headers=headers, json=body, timeout=prov["timeout"])
   290→            resp.raise_for_status()
   291→            data = resp.json()
   292→
   293→            if prov["format"] == "openai":
   294→                reply = data["choices"][0]["message"]["content"]
   295→                t_in = data.get("usage", {}).get("prompt_tokens", 0)
   296→                t_out = data.get("usage", {}).get("completion_tokens", 0)
   297→            else:
   298→                reply = data["content"][0]["text"]
   299→                t_in = data.get("usage", {}).get("input_tokens", 0)
   300→                t_out = data.get("usage", {}).get("output_tokens", 0)
   301→
   302→            costs = TOKEN_COSTS.get(prov_name, {"input": 1, "output": 1})
   303→            cost = t_in * costs["input"] / 1_000_000 + t_out * costs["output"] / 1_000_000
   304→            now = datetime.utcnow().isoformat()
   305→            _query("INSERT INTO token_usage (provider,model,tokens_in,tokens_out,cost,dept,created) VALUES (?,?,?,?,?,?,?)",
   306→                   (prov_name, use_model, t_in, t_out, round(cost, 6), dept, now))
   307→            _query("INSERT INTO audit_log (action,details,created) VALUES (?,?,?)",
   308→                   ("llm_call", json.dumps({"dept": dept, "provider": prov_name, "model": use_model, "tokens_in": t_in, "tokens_out": t_out}), now))
   309→            log.info(f"LLM ok: provider={prov_name} model={use_model} dept={dept} t_in={t_in} t_out={t_out}")
   310→            return {"reply": reply, "provider": prov_name, "model": use_model, "tokens_in": t_in, "tokens_out": t_out}
   311→        except Exception as e:
   312→            last_err = str(e)
   313→            log.warning(f"LLM fail: provider={prov_name} err={last_err}")
   314→    return {"reply": f"[Oshibka] Vse providery nedostupny: {last_err}", "provider": None, "model": None, "tokens_in": 0, "tokens_out": 0}

   316→# ──────────────────────────── TELEGRAM HELPERS ────────────────────────────
   317→def _tg_send_message(chat_id, text, parse_mode=None):
   318→    """Send a text message via Telegram Bot API."""
   319→    if not TELEGRAM_BOT_TOKEN:
   320→        log.warning("TELEGRAM_BOT_TOKEN not set, cannot send message")
   321→        return None
   322→    url = f"https://api.telegram.org/bot{TELEGRAM_BOT_TOKEN}/sendMessage"
   323→    payload = {"chat_id": chat_id, "text": text}
   324→    if parse_mode:
   325→        payload["parse_mode"] = parse_mode
   326→    try:
   327→        resp = http_requests.post(url, json=payload, timeout=30)
   328→        resp.raise_for_status()
   329→        data = resp.json()
   330→        if data.get("ok"):
   331→            return data["result"].get("message_id")
   332→    except Exception as e:
   333→        log.error(f"Telegram sendMessage failed: {e}")
   334→    return None

   336→def _tg_send_voice(chat_id, voice_path):
   337→    """Send a voice file via Telegram Bot API using multipart upload."""
   338→    if not TELEGRAM_BOT_TOKEN:
   339→        log.warning("TELEGRAM_BOT_TOKEN not set, cannot send voice")
   340→        return None
   341→    url = f"https://api.telegram.org/bot{TELEGRAM_BOT_TOKEN}/sendVoice"
   342→    try:
   343→        with open(voice_path, "rb") as f:
   344→            resp = http_requests.post(url, data={"chat_id": chat_id}, files={"voice": f}, timeout=60)
   345→        resp.raise_for_status()
   346→        data = resp.json()
   347→        if data.get("ok"):
   348→            return data["result"].get("message_id")
   349→    except Exception as e:
   350→        log.error(f"Telegram sendVoice failed: {e}")
   351→    return None

   614→@app.post("/tts")
   615→def tts_endpoint():
   616→    """Text-to-speech via edge-tts, send voice to Telegram."""
   617→    d = request.get_json(force=True)
   618→    text = d.get("text", "")
   619→    voice = d.get("voice", "ru-RU-DmitryNeural")
   620→    chat_id = d.get("chat_id", "")
   621→
   622→    if not text:
   623→        return jsonify({"status": "error", "message": "text is required"}), 400
   624→    if not chat_id:
   625→        return jsonify({"status": "error", "message": "chat_id is required"}), 400
   626→
   627→    timestamp = int(time.time())
   628→    tmp_path = f"/tmp/tts_{timestamp}_{uuid.uuid4().hex[:6]}.mp3"
   629→
   630→    try:
   631→        # Run edge-tts subprocess
   632→        cmd = [
   633→            "edge-tts",
   634→            "--voice", voice,
   635→            "--text", text,
   636→            "--write-media", tmp_path
   637→        ]
   638→        result = subprocess.run(cmd, capture_output=True, text=True, timeout=60)
   639→        if result.returncode != 0:
   640→            log.error(f"edge-tts failed: {result.stderr}")
   641→            _log_audit("tts_error", {"voice": voice, "error": result.stderr[:500]})
   642→            return jsonify({"status": "error", "message": "edge-tts synthesis failed", "detail": result.stderr[:500]}), 500
   643→
   644→        # Send voice to Telegram
   645→        msg_id = _tg_send_voice(chat_id, tmp_path)
   646→
   647→        _log_audit("tts", {"chat_id": chat_id, "voice": voice, "text_len": len(text), "tg_msg_id": msg_id})
   648→
   649→        return jsonify({"status": "ok", "msg_id": msg_id})
   650→
   651→    except FileNotFoundError:
   652→        _log_audit("tts_error", {"error": "edge-tts not installed"})
   653→        return jsonify({"status": "error", "message": "edge-tts binary not found"}), 500
   654→    except subprocess.TimeoutExpired:
   655→        _log_audit("tts_error", {"error": "timeout"})
   656→        return jsonify({"status": "error", "message": "TTS synthesis timeout"}), 500
   657→    except Exception as e:
   658→        log.error(f"TTS error: {e}")
   659→        _log_audit("tts_error", {"error": str(e)[:500]})
   660→        return jsonify({"status": "error", "message": str(e)}), 500
   661→    finally:
   662→        # Clean up temp file
   663→        try:
   664→            if os.path.exists(tmp_path):
   665→                os.remove(tmp_path)
   666→        except Exception:
   667→            pass

   670→#  MEGA-3: STT (WHISPER) ENDPOINT
   671→# ══════════════════════════════════════════════════════════════════════
   672→
   673→@app.post("/stt")
   674→def stt_endpoint():
   675→    """Speech-to-text via Whisper (Groq preferred, OpenAI fallback)."""
   676→    d = request.get_json(force=True)
   677→    file_id = d.get("file_id", "")
   678→
   679→    if not file_id:
   680→        return jsonify({"status": "error", "message": "file_id is required"}), 400
   681→
   682→    provider = WHISPER_PROVIDER
   683→    audio_data = None
   684→
   685→    try:
   686→        # Step 1: Get file path from Telegram
   687→        file_path = _tg_get_file(file_id)
   688→        if not file_path:
   689→            _log_audit("stt_error", {"error": "could not get file path from Telegram", "file_id": file_id})
   690→            return jsonify({"status": "error", "message": "Failed to get file from Telegram"}), 400
   691→
   692→        # Step 2: Download audio from Telegram
   693→        audio_data = _tg_download_file(file_path)
   694→        if not audio_data:
   695→            _log_audit("stt_error", {"error": "could not download file from Telegram", "file_path": file_path})
   696→            return jsonify({"status": "error", "message": "Failed to download audio from Telegram"}), 400
   697→
   698→        # Step 3: Transcribe via Whisper
   699→        if provider == "groq":
   700→            api_key = os.environ.get("GROQ_API_KEY", WHISPER_API_KEY)
   701→            url = "https://api.groq.com/openai/v1/audio/transcriptions"
   702→            model = "whisper-large-v3"
   703→        else:
   704→            api_key = os.environ.get("OPENAI_API_KEY", WHISPER_API_KEY)
   705→            url = "https://api.openai.com/v1/audio/transcriptions"
   706→            model = "whisper-1"
   707→
   708→        # Prepare multipart form
   709→        files = {
   710→            "file": ("audio.ogg", bytes(audio_data), "audio/ogg"),
   711→        }
   712→        data = {"model": model}
   713→        headers = {"Authorization": f"Bearer {api_key}"}
   714→
   715→        resp = http_requests.post(url, headers=headers, files=files, data=data, timeout=60)
   716→        resp.raise_for_status()
   717→        resp_data = resp.json()
   718→        transcribed_text = resp_data.get("text", "")
   719→
   720→        _log_audit("stt", {"file_id": file_id, "provider": provider, "model": model, "text_len": len(transcribed_text)})
   721→
   722→        return jsonify({"status": "ok", "text": transcribed_text, "provider": provider})
   723→
   724→    except Exception as e:
   725→        log.error(f"STT error (provider={provider}): {e}")
   726→        # Try fallback provider if primary failed
   727→        if provider == "groq" and audio_data:
   728→            try:
   729→                fallback_key = os.environ.get("OPENAI_API_KEY", "")
   730→                if not fallback_key:

   856→def _collect_metrics_for_period(start_iso, end_iso=None):
   857→    """Collect metrics for a given time period."""
   858→    conditions = "created >= ?"
   859→    params = [start_iso]
   860→    if end_iso:
   861→        conditions += " AND created < ?"
   862→        params.append(end_iso)
   863→
   864→    tasks = _query(f"SELECT dept, COUNT(*) as c FROM dialog_history WHERE {conditions} AND role='user' GROUP BY dept", params, fetch=True)
   865→    reports = _query(f"SELECT dept, COUNT(*) as c FROM miniapp_responses WHERE {conditions} GROUP BY dept", params, fetch=True)
   866→    corrections = _query(f"SELECT dept, COUNT(*) as c FROM adapt_corrections WHERE {conditions} GROUP BY dept", params, fetch=True)
   867→    ratings = _query(f"SELECT dept, COALESCE(AVG(rating),0) as avg_r, COUNT(*) as c FROM dept_metrics WHERE {conditions} GROUP BY dept", params, fetch=True)
   868→    tokens = _query(f"SELECT dept, COALESCE(SUM(tokens_in),0) as ti, COALESCE(SUM(tokens_out),0) as to2, COALESCE(SUM(cost),0) as cost FROM token_usage WHERE {conditions} GROUP BY dept", params, fetch=True)
   869→
   870→    return {
   871→        "tasks": {r["dept"]: r["c"] for r in tasks},
   872→        "reports": {r["dept"]: r["c"] for r in reports},
   873→        "corrections": {r["dept"]: r["c"] for r in corrections},
   874→        "ratings": {r["dept"]: {"avg": r["avg_r"], "count": r["c"]} for r in ratings},
   875→        "tokens": {r["dept"]: {"in": r["ti"], "out": r["to2"], "cost": r["cost"]} for r in tokens},
   876→    }
   877→
   878→def _format_metrics_text(metrics):
   879→    """Format metrics dict as readable text for LLM prompt."""
   880→    lines = []
   881→    lines.append("METRICS:")
   882→    all_depts = set(list(metrics["tasks"].keys()) + list(metrics["reports"].keys()) + list(metrics["corrections"].keys()) + list(metrics["ratings"].keys()))
   883→    for dept in sorted(all_depts):
   884→        name = DEPT_NAMES.get(dept, dept)
   885→        parts = [f"{name}:"]
   886→        if dept in metrics["tasks"]:
   887→            parts.append(f"tasks={metrics['tasks'][dept]}")
   888→        if dept in metrics["reports"]:
   889→            parts.append(f"reports={metrics['reports'][dept]}")
   890→        if dept in metrics["corrections"]:
   891→            parts.append(f"corrections={metrics['corrections'][dept]}")
   892→        if dept in metrics["ratings"]:
   893→            r = metrics["ratings"][dept]
   894→            parts.append(f"avg_rating={r['avg']:.1f} (n={r['count']})")
   895→        if dept in metrics["tokens"]:
   896→            t = metrics["tokens"][dept]
   897→            parts.append(f"tokens={t['in']+t['out']} cost=${t['cost']:.2f}")
   898→        lines.append("  " + " ".join(parts))
   899→    return "\n".join(lines)

  1395→def _send_chat_action_loop(chat_id, action="typing", interval=5):
  1396→    """Send typing/action indicator to Telegram in a loop. Returns a stop event."""
  1397→    stop = threading.Event()
  1398→
  1399→    def _loop():
  1400→        while not stop.is_set():
  1401→            try:
  1402→                _tg_send_chat_action(chat_id, action)
  1403→            except Exception:
  1404→                pass
  1405→            stop.wait(interval)
  1406→
  1407→    t = threading.Thread(target=_loop, daemon=True)
  1408→    t.start()
  1409→    return stop
  1410→
  1411→def _tg_send_chat_action(chat_id, action):
  1412→    """Send a chat action to Telegram (typing, upload_photo, etc.)."""
  1413→    if not TELEGRAM_BOT_TOKEN:
  1414→        return
  1415→    url = f"https://api.telegram.org/bot{TELEGRAM_BOT_TOKEN}/sendChatAction"
  1416→    try:
  1417→        http_requests.post(url, json={"chat_id": chat_id, "action": action}, timeout=10)
  1418→    except Exception:
  1419→        pass
